function escapeHTML(value){
  return String(value).replace(/[&<>"']/g, character => ({
    "&":"&amp;",
    "<":"&lt;",
    ">":"&gt;",
    '"':"&quot;",
    "'":"&#39;"
  }[character]));
}

/* 圖示載入失敗時，仍可使用原本符號 */
function iconHTML(symbol){
  if(typeof illustratedIcon === "function"){
    return illustratedIcon(symbol);
  }
  return escapeHTML(symbol);
}

  if(!(savedPhone || SUPERVISOR_PHONE_NUMBER)){

    openSettings();
  }

$("#phoneForm").onsubmit = event => {
  event.preventDefault();

  const value = $("#phone").value.replace(/[\s()-]/g, "");

  if(value && !/^\+?\d{6,15}$/.test(value)){
    $("#error").textContent =
      "請輸入有效的電話號碼（6～15碼數字）。";
    return;
  }

  try{
    localStorage.setItem("supervisorPhone", value);
    savedPhone = value;
  }catch{
    $("#error").textContent =
      "無法儲存，請確認瀏覽器允許儲存設定。";
    return;
  }

  updatePhone();
  $("#dialog").close();
};

function card(icon, title, subtitle, href){
  const isCategory = /^#category\/\d+$/.test(href);

  const picture = isCategory
    ? ""
    : `<span class="icon" aria-hidden="true">${iconHTML(icon)}</span>`;

  return `
    <a
      class="card ${isCategory ? "category-text-card" : ""}"
      href="${href}"
    >
      ${picture}
      <span class="cardtext">
        <b>${escapeHTML(title)}</b>
        ${subtitle ? `<small>${escapeHTML(subtitle)}</small>` : ""}
      </span>
    </a>
  `;
}

  }else{
    location.hash = "";
  }
}

    </button>
  `;
}

/* 按鈕文字自動放大 */
function fitCardLabels(){
  const canvas = document.createElement("canvas");
  const context = canvas.getContext("2d");

  if(!context) return;

  document.querySelectorAll(".card b").forEach(label => {
    const button = label.closest(".card");

    if(button.closest(".uniform-events")){
      label.style.removeProperty("font-size");
      label.style.removeProperty("white-space");
      return;
    }

    const textBox = label.parentElement;
    const width = textBox.clientWidth - 2;

    if(width <= 0) return;

    const style = getComputedStyle(label);
    context.font = "700 100px " + style.fontFamily;

    const measured = context.measureText(label.textContent).width;
    const buttonStyle = getComputedStyle(button);

    const availableHeight =
      button.clientHeight -
      parseFloat(buttonStyle.paddingTop) -
      parseFloat(buttonStyle.paddingBottom);

    const limit = document.body.classList.contains("category-screen")
      ? Math.min(64, availableHeight / 1.3)
      : 32;

    const fitted = Math.floor(
      Math.min(limit, width / Math.max(measured, 1) * 100)
    );

    label.style.fontSize = Math.max(22, fitted) + "px";
    label.style.whiteSpace = fitted >= 22 ? "nowrap" : "normal";
  });
}

function categoryCards(){
  return categories.map((category, index) => card(
    categoryIcons[index],
    category[0],
    categoryDescriptions[index],
    `#category/${index}`
  )).join("");
}

    const item = categories[categoryIndex][2][eventIndex];

    html = back() + `
      <div class="description">
        ${escapeHTML(item[1])}
      </div>

      ${item[2] ? `
        <aside class="warning">
          <b>⚠ 重要提醒</b>
          ${escapeHTML(item[2])}
        </aside>
      ` : ""}
    `;

    if(item[4]){
      html += `
        <section class="procedure">
          <h2>處理步驟</h2>
          <ol class="steps">
            ${item[4].map(step => `
              <li>
                <span>
                  <span class="step-symbol" aria-hidden="true">
                    ${actionIcon(step)}
                  </span>
                  ${escapeHTML(step)}
                </span>
              </li>
            `).join("")}
          </ol>
        </section>
      `;
    }

    html += item[3].map(([level, situation, action]) => `
      <section class="level ${level}">
        <h2 class="situation">
          <span aria-hidden="true">
            ${{r:"🔴", y:"🟡", g:"🟢"}[level]}
          </span>
          ${escapeHTML(situation)}
        </h2>

        <p class="response">
          <span aria-hidden="true">${actionIcon(action)}</span>
          ${escapeHTML(action)}
        </p>
      </section>
    `).join("");

  }else if(isCategory){
    const category = categories[categoryIndex];
    const vertical = [2,3,4,5].includes(categoryIndex);

    const rows = vertical
      ? category[2].length
      : Math.ceil(category[2].length / 2);

    html = back() + `
      <div
        class="list event-grid
          ${vertical ? "vertical-events" : ""}
          ${categoryIndex === 0 ? "uniform-events" : ""}"
        style="--event-rows:${rows}"
        aria-label="${escapeHTML(category[0])}事件列表"
      >
        ${category[2].map((item, index) => card(
          eventIcons[categoryIndex][index],
          item[0],
          "",
          `#event/${categoryIndex}/${index}`
        )).join("")}
      </div>
    `;

  }else{
    html = `
      <section class="hero">
        <strong>🚨 請立即撥打 119</strong>
        <p>若意識不清楚，無正常呼吸或情況危急時</p>
        <div class="notes">
          ※ 保持冷靜，確保自身與個案安全。<br>
          ※ 遇到異常事件，請主動通報，切勿隱瞞。
        </div>
      </section>

      <label class="search">
        <span aria-hidden="true">⌕</span>
        <input
          id="search"
          type="search"
          placeholder="搜尋事件，例如：跌倒"
          aria-label="搜尋事件"
        >
      </label>

      <div class="sectionlabel">
        <h2 id="listTitle">請選擇事件分類</h2>
        <span id="count">6 大類 · 25 個事件</span>
      </div>

      <div class="list" id="results">
        ${categoryCards()}
      </div>
    `;
  }

  document.body.classList.toggle("category-screen", isCategory);
  $(".brand").hidden = isCategory || isEvent;

  $("#app").innerHTML = `
    <div class="page">
      ${html}
      <p class="footer-note">
        先確保安全，再依情況處理與通報。
      </p>
    </div>
  `;

  requestAnimationFrame(fitCardLabels);
  window.scrollTo({top:0, behavior:"instant"});

  if(isCategory || isEvent){
    $("#app .back")?.focus({preventScroll:true});
  }

  const search = $("#search");

  if(search){
    search.oninput = () => {
      const query = search.value.trim();

      if(!query){
        $("#results").innerHTML = categoryCards();
        $("#listTitle").textContent = "請選擇事件分類";
        $("#count").textContent = "6 大類 · 25 個事件";
        requestAnimationFrame(fitCardLabels);
        return;
      }

      const matches = [];

      categories.forEach((category, categoryIndex) => {
        category[2].forEach((item, eventIndex) => {
          if((item[0] + item[1]).includes(query)){
            matches.push(card(
              eventIcons[categoryIndex][eventIndex],
              item[0],
              category[0],
              `#event/${categoryIndex}/${eventIndex}`
            ));
          }
        });
      });

      $("#listTitle").textContent = "搜尋結果";
      $("#count").textContent = matches.length + " 個事件";

      $("#results").innerHTML = matches.join("") || `
        <div class="empty">
          找不到符合的事件，請換個關鍵字，或從分類查看。
        </div>
      `;

      requestAnimationFrame(fitCardLabels);
    };
  }
}

window.addEventListener("resize", () => {
  requestAnimationFrame(fitCardLabels);
});

document.fonts?.ready.then(() => {
  requestAnimationFrame(fitCardLabels);
});

updatePhone();
render();
